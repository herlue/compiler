namespace main;
use io;
use math;

record Point {
  i32 x;
  i32 y;
  record Origin {
    i32 x;
    i32 y;
  };
};

u32 (Point p).distanceTo(i32 x, i32 y) {
  u32 distX = math::abs(p.x - x);
  u32 distY = math::abs(p.y - y);

  return math::sqrt(distX * distX + distY * distY);
}

i32 main(i32 argc, byte*[] argv) {
  Point p = Point {
    x: 16,
    y: 32
  };


  return 0;
}
